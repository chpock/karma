exec bash -x ./test.sh &
! exists karma-state
exec managed-karma karma-state karma.pid 5s 2s karma --log.timestamp=true --pid-file=karma.pid --listen.address=127.0.0.1 --listen.port=8062
! stdout .
stderr 'time=[0-9]+-[0-9]+-[0-9]+T[0-9]+:[0-9]+:[0-9]+.* level=INFO msg="Version: dev"'
stderr 'time=[0-9]+-[0-9]+-[0-9]+T[0-9]+:[0-9]+:[0-9]+.* level=INFO msg="Pulling latest alerts and silences from Alertmanager"'
wait
